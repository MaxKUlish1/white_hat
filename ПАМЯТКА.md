# white_hat — в кабинете (всё в ~/Desktop/white_hat, в систему ничего не ставилось)

## САМОЕ ПРОСТОЕ: мастер ведёт по шагам и ничего не даёт забыть
    whm   ->  g          (или: cd ~/Desktop/white_hat && source env.sh && wh go)

## Старт (2 команды)
    cd ~/Desktop/white_hat && source env.sh
    wh shell                 # терминал под надзором: КАЖДАЯ твоя команда -> журнал, сессия -> logs/

## Перед входом в кабинет
    wh doctor                # всё ли готово: 11 инструментов, ключ, базы, интернет, надзор, журнал
    wh demo                  # показ офлайн: сторож режет nmap/чужой IP/sudo/запись в /etc; подделка журнала ловится

## Порядок работы
    wh start <IP> <логин>    # IP от админа -> область (scope.txt) + клятва
    wh key                   # твой ключ SSH: публичную строку отдать админу (без паролей)
    wh fp <IP>               # отпечаток ключа сервера — СВЕРИТЬ с админом до входа
    wh versions <IP>         # версии ПО + гигиена (SSH root/пароли, перезагрузка, автообновления, учётки, фаервол) -> отчёт out/
    wh sshaudit <IP>         # настройки SSH сервера (без входа)
    wh tls <IP>              # HTTPS: протоколы, шифры, сертификат
    wh web <IP>              # что за веб: код ответа, сервер, технологии
    wh check --offline <файл> # сверка БЕЗ интернета (trivy, локальная база, ~5 с) — включается и сама, если сети нет
    wh cve <CVE-ID>          # что за уязвимость — описание и оценка офлайн
    wh plan <IP>             # план работ для админа (показать и согласовать ДО работы)
    wh about                 # схема «как работает» в браузере
    wh report                # итоговый файл out/ИТОГ_*.md + хэш журнала -> отдать админу
    wh ssh <IP>              # обычный вход (сессия записывается); если дали ПАРОЛЬ — спросит пароль сам
    wh winversions <IP>      # если сервер Windows (OpenSSH): список ПО и обновлений, только чтение
    # Windows без SSH: скопировать version.ps1 на сервер (с разрешения) и запустить:
    #   powershell -NoProfile -ExecutionPolicy Bypass -File version.ps1 ; файл versions_*.txt -> wh check <файл>
    wh nuclei <IP>           # ТОЛЬКО по отдельному разрешению, безопасный режим
    wh code <папка>          # разбор кода у себя (semgrep, gitleaks, trivy)
    wh log | wh undo | wh verify

## ИИ под надзором
    claude                   # запускать ТОЛЬКО из этой папки
    - сторож: каждое действие ИИ ДО выполнения — адреса только из scope.txt, запись только в папке проекта,
      закрытый ключ ИИ не читает; sudo/службы/установка/nmap/подбор/выгрузка баз — запрещены всегда
    - капсула после каждого действия (ДЕЛАЛ -> ПОЛУЧИЛ) + автокоммит в git; откат: wh undo
    - журнал journal.tsv — цепочка хэшей; wh verify показывает, цел ли журнал
